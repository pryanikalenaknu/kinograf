# Українські драми від 1991 року
SELECT ?filmLabel ?year WHERE {
  ?film wdt:P31 wd:Q11424 ;
        wdt:P495 wd:Q212 ;
        wdt:P136 wd:Q130232 ;
        wdt:P577 ?date .
  BIND(YEAR(?date) AS ?year)
  FILTER(?year >= 1991)
  SERVICE wikibase:label { bd:serviceParam wikibase:language "uk". }
}
ORDER BY DESC(?year)
LIMIT 50

PREFIX wd: <http://www.wikidata.org/entity/>
PREFIX wdt: <http://www.wikidata.org/prop/direct/>
# Українські драми від 1991 року
SELECT ?filmLabel ?year WHERE {
  ?film wdt:P31 wd:Q11424 ;
        wdt:P495 wd:Q212 ;
        wdt:P136 wd:Q130232 ;
        wdt:P577 ?date .
  BIND(YEAR(?date) AS ?year)
  FILTER(?year >= 1991)
  SERVICE wikibase:label { bd:serviceParam wikibase:language "uk". }
}
ORDER BY DESC(?year)
